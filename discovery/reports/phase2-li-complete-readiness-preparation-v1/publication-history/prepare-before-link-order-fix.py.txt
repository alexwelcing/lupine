"""Explicit public evidence copy; never imports candidate code or reads actual input."""
from datetime import datetime, timezone
from hashlib import sha256
import json
from pathlib import Path, PurePosixPath
import re

ROOT = Path(__file__).resolve().parents[2]
LANE = Path(__file__).resolve().parent
TARGET = Path('reports/phase2-li-complete-readiness-preparation-v1')
OUT = LANE / 'promotion' / TARGET

def digest(b):
    return sha256(b).hexdigest()

def write_json(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, sort_keys=True, ensure_ascii=True) + '\n')

inputs = []

def add(source, destination, expected=None):
    source = PurePosixPath(source)
    assert not source.is_absolute() and '..' not in source.parts
    assert not any(p in ('__pycache__', 'isolated-python-cache', 'actual') for p in source.parts)
    assert not str(source).endswith(('.pyc', '.zip', '.whl'))
    path = ROOT / source
    assert path.is_file() and not path.is_symlink(), source
    raw = path.read_bytes()
    raw.decode('utf-8')
    if expected is not None:
        assert digest(raw) == expected['sha256'], source
        if 'bytes' in expected:
            assert len(raw) == expected['bytes'], source
    inputs.append({'source': str(source), 'logical_destination': str(destination),
                   'source_bytes': len(raw), 'source_sha256': digest(raw), 'raw': raw})

for name, dest, inventory_sha, handoff_sha, count in [
    ('phase2-li-complete-readiness-candidate-v1', 'author',
     '4e98dbe47a148cd0bb74df2c8a158f1427f4d6475a9fda53852f58fefa04c9c7',
     '9a3abc91', 58),
    ('phase2-li-complete-readiness-review-v1', 'independent',
     'db721cc93191614a5dc6f13381c0bfbd251b0857e409be1418dcd46e2e34a032',
     '9cb7599fce4c9e7a82c5c6150fd775f03cee4d64d9562a51c4d08f696e076b47', 40),
]:
    base = Path('.cache') / name
    invraw = (ROOT / base / 'inventory.json').read_bytes()
    hraw = (ROOT / base / 'handoff.json').read_bytes()
    assert digest(invraw) == inventory_sha
    assert digest(hraw).startswith(handoff_sha)
    inventory = json.loads(invraw)
    handoff = json.loads(hraw)
    assert len(inventory['files']) == count and handoff['inventory_sha256'] == inventory_sha
    for rel, expected in sorted(inventory['files'].items()):
        add(base / rel, Path(dest) / rel, expected)
    add(base / 'inventory.json', Path(dest) / 'inventory.json', {'sha256': inventory_sha})
    add(base / 'handoff.json', Path(dest) / 'handoff.json', {'sha256': digest(hraw)})

OUT.mkdir(parents=True, exist_ok=False)
plan = {x['logical_destination'] for x in inputs}
original_link_findings = []
entries = []
for item in inputs:
    raw = item['raw']
    logical = item['logical_destination']
    reason = []
    if logical.endswith('.log'):
        reason.append('Preserve ignored log bytes in an explicitly staged JSON artifact.')
    if logical.endswith('.diff'):
        reason.append('Preserve exact historical diff including whitespace-bearing context lines.')
    if any(line.rstrip(b' \t') != line for line in raw.splitlines()):
        reason.append('Preserve original trailing whitespace without weakening publication whitespace checks.')
    if raw.endswith(b'\n\n'):
        reason.append('Preserve original extra final blank lines.')
    if logical.endswith('.md'):
        for link in re.findall(r'\]\(([^)]+)\)', raw.decode('utf-8')):
            if '://' in link or link.startswith('#'):
                continue
            target = link.split('#')[0].strip('<>')
            proposed = (OUT / logical).parent / target
            normalized = proposed.resolve().relative_to(OUT.resolve()) if proposed.resolve().is_relative_to(OUT.resolve()) else None
            if normalized is None or str(normalized) not in plan:
                original_link_findings.append({'source': item['source'], 'link': link})
                reason.append('Preserve historical Markdown whose relative links do not resolve after relocation.')
    destination = logical + '.json' if reason else logical
    path = OUT / destination
    path.parent.mkdir(parents=True, exist_ok=True)
    if reason:
        write_json(path, {'schema': 'lupine.lossless_utf8_publication_wrapper.v1',
                          'original_source': item['source'], 'original_relative_path': logical,
                          'original_bytes': len(raw), 'original_sha256': digest(raw),
                          'encoding': 'utf-8', 'reasons': list(dict.fromkeys(reason)),
                          'text': raw.decode('utf-8')})
        reconstructed = json.loads(path.read_bytes())['text'].encode('utf-8')
    else:
        path.write_bytes(raw)
        reconstructed = path.read_bytes()
    assert reconstructed == raw
    entry = {key: value for key, value in item.items() if key != 'raw'}
    entry.update(destination=str(TARGET / destination), destination_relative=destination,
                 destination_bytes=path.stat().st_size, destination_sha256=digest(path.read_bytes()),
                 transformation='lossless_utf8_json_wrapper' if reason else 'exact_copy',
                 reasons=list(dict.fromkeys(reason)))
    entries.append(entry)
